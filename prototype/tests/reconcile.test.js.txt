import test from "node:test";
import assert from "node:assert/strict";
import { JSDOM } from "jsdom";
import { reconcileChildren } from "../prototype/modules/ui/reconcile.js";
function setup(html) {
  const dom = new JSDOM(`<main>${html}</main>`);
  const document = dom.window.document;
  return {
    dom,
    document,
    root: document.querySelector("main"),
    update(html) {
      const fresh = document.createElement("div");
      fresh.innerHTML = html;
      reconcileChildren(document.querySelector("main"), [...fresh.childNodes]);
    },
  };
}
test("updates preserve group, tab, marker and listeners", () => {
  const f = setup(
    '<section class="group" data-group-id="a"><span class="marker"></span><div class="tab-row" data-tab-id="1"><span>Old</span></div></section>',
  );
  const group = f.root.firstChild,
    marker = group.firstChild,
    tab = group.lastChild;
  let clicks = 0;
  tab.addEventListener("click", () => clicks++);
  f.update(
    '<section class="group active" data-group-id="a"><span class="marker"></span><div class="tab-row" data-tab-id="1"><span>New</span></div></section>',
  );
  assert.equal(f.root.firstChild, group);
  assert.equal(group.firstChild, marker);
  assert.equal(group.lastChild, tab);
  assert.equal(tab.textContent, "New");
  tab.click();
  assert.equal(clicks, 1);
  f.dom.window.close();
});
test("group creation drafts survive unrelated background updates", () => {
  const f = setup(
    '<button class="create-group" hidden>Add</button><form class="new-group-form"><input value=""></form>',
  );
  const input = f.root.querySelector("input");
  input.value = "Draft";
  input.focus();
  f.update(
    '<button class="create-group">Add</button><form class="new-group-form" hidden><input value=""></form>',
  );
  assert.equal(f.document.activeElement, input);
  assert.equal(input.value, "Draft");
  assert.equal(input.closest("form").hidden, false);
  assert.equal(f.root.querySelector("button").hidden, true);
  f.dom.window.close();
});
test("reorder retains keyed group nodes", () => {
  const f = setup(
    '<section class="group" data-group-id="a">A</section><section class="group" data-group-id="b">B</section>',
  );
  const a = f.root.firstChild,
    b = f.root.lastChild;
  f.update(
    '<section class="group" data-group-id="b">B</section><section class="group" data-group-id="a">A</section>',
  );
  assert.equal(f.root.firstChild, b);
  assert.equal(f.root.lastChild, a);
  f.dom.window.close();
});
